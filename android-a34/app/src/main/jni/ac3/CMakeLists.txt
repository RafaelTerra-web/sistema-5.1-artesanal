set(FFMPEG_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../third-party/ffmpeg-arm64")
if(NOT EXISTS "${FFMPEG_ROOT}/lib/libavcodec.a")
    message(FATAL_ERROR "Build the pinned LGPL AC-3 dependency first: scripts/build-ac3-dependency.ps1 or scripts/build-ac3-dependency.sh")
endif()
add_library(sistema51_ac3 SHARED ffmpeg_ac3_jni.c)
target_include_directories(sistema51_ac3 PRIVATE "${FFMPEG_ROOT}/include")
target_compile_options(sistema51_ac3 PRIVATE -Wall -Wextra -Werror -std=c11)
target_link_libraries(sistema51_ac3 PRIVATE "${FFMPEG_ROOT}/lib/libavcodec.a" "${FFMPEG_ROOT}/lib/libavutil.a" m atomic android mediandk)
target_link_options(sistema51_ac3 PRIVATE "-Wl,-z,max-page-size=16384")
